function(snr_target target)
    set_target_properties(${target} PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF OBJC_STANDARD 23 OBJC_STANDARD_REQUIRED ON OBJC_EXTENSIONS OFF C_VISIBILITY_PRESET hidden OBJC_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON)
    target_compile_features(${target} PUBLIC c_std_23)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wshadow -Wconversion -Wsign-conversion -ffp-contract=off -Wstrict-prototypes -Wmissing-prototypes -Wpointer-arith -Wcast-qual)
    if(SNRENDER_WERROR)
        target_compile_options(${target} PRIVATE -Werror)
    endif()
    get_target_property(target_type ${target} TYPE)
    if(WIN32 AND target_type STREQUAL "EXECUTABLE")
        # Stage runtime dependencies beside every executable so direct runs,
        # CTest and executables launched by Python use the same DLL search path.
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND}
                "-DSNR_RUNTIME_DLLS=$<TARGET_RUNTIME_DLLS:${target}>"
                "-DSNR_RUNTIME_DESTINATION=$<TARGET_FILE_DIR:${target}>"
                -P "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/cmake/CopyRuntimeDLLs.cmake"
            VERBATIM)
    endif()
endfunction()
